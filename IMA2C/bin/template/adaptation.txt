#define CHECK_CODE(msg, code) \
    if(code == NO_ERROR) \
    { \
        printf("%s : SUCCESSFUL\n" ,msg); \
    }\
    else \
    { \
        printf("%s : FAILED %s\n" ,msg, codeToStr(code)); \
    }

#define CHECK_VALIDCODE(msg, validCode, ret) \
    if((validCode != VALID) && (ret != NO_ERROR)) \
    { \
        printf("%s FAILED %s, %d\n", msg, codeToStr(ret), validCode); \
    } \
    else \
    { \
        printf("%s SUCCESSFUL %s, %d\n", msg, codeToStr(ret), validCode); \
    }
